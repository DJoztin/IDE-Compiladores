main() {
    /* G5-4 (repetitivo): el mayor y el menor de N enteros */
    int n, i, x, mayor, menor;
    do {
        cin("Cuantos numeros (N >= 1)", n);
        if (n < 1) then
            cout "N debe ser 1 o mayor";
        end;
    } until (n >= 1);
    i = 0;
    while (i < n) {
        cin("Numero entero", x);
        if (i == 0) then
            mayor = x;
            menor = x;
        else
            if (x > mayor) then
                mayor = x;
            end;
            if (x < menor) then
                menor = x;
            end;
        end;
        i++;
    };
    cout "Mayor", mayor;
    cout "Menor", menor;
}
