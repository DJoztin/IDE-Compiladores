main() {
    /* G2-2, G9-2 (selectivo): T = h / 4 + 40 (grados Fahrenheit) */
    float h, t;
    cin("Sonidos del grillo por minuto (h >= 0)", h);
    if (h >= 0) then
        t = h / 4 + 40;
        cout "Temperatura (F)", t;
    else
        cout "El numero de sonidos no puede ser negativo";
    end;
}
