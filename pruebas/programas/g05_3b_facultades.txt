main() {
    /* G5-3 (selectivo): Economia: semestre >= 6 y promedio >= 8.8;
       Computacion: semestre > 6 y promedio > 8.5;
       Administracion y Contabilidad: semestre > 5 y promedio > 8.5.
       Si es aceptado imprime matricula, carrera y Aceptado = 1; si no, 0. */
    int matricula, carrera, semestre, acepta;
    float prom;
    cin("Matricula del alumno", matricula);
    cin("Carrera (1 Economia, 2 Computacion, 3 Administracion, 4 Contabilidad)", carrera);
    cin("Semestre (1 a 12)", semestre);
    cin("Promedio (0 a 10)", prom);
    if (carrera < 1 || carrera > 4 || semestre < 1 || semestre > 12
        || prom < 0 || prom > 10) then
        cout "Datos invalidos: revisa carrera (1-4), semestre (1-12) y promedio (0-10)";
    else
        acepta = 0;
        if (carrera == 1) then
            if (semestre >= 6 && prom >= 8.8) then acepta = 1; end;
        else
            if (carrera == 2) then
                if (semestre > 6 && prom > 8.5) then acepta = 1; end;
            else
                if (semestre > 5 && prom > 8.5) then acepta = 1; end;
            end;
        end;
        if (acepta == 1) then
            cout "Matricula", matricula;
            cout "Carrera", carrera;
            cout "Aceptado", 1;
        else
            cout "Aceptado (0 = no)", 0;
        end;
    end;
}
