main() {
    /* G4-4: N ventas. Imprime cuantas y cuanto suman las de $200 o menos,
       las mayores a $200 e inferiores a $400, y las de $400 o mas. */
    int n, i, c1, c2, c3;
    float v, s1, s2, s3;
    cin(n);
    c1 = 0; c2 = 0; c3 = 0;
    s1 = 0; s2 = 0; s3 = 0;
    i = 1;
    while (i <= n) {
        cin(v);
        if (v <= 200) then
            c1++;
            s1 = s1 + v;
        else
            if (v < 400) then
                c2++;
                s2 = s2 + v;
            else
                c3++;
                s3 = s3 + v;
            end;
        end;
        i++;
    };
    cout c1; cout s1;
    cout c2; cout s2;
    cout c3; cout s3;
}
