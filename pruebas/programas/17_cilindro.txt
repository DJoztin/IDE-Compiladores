main() {
    /* G5-1, G9-1 (secuencial): Volumen = pi * radio^2 * altura ;
       Area = 2 * pi * radio * altura */
    float r, h, area, vol;
    cin("Radio del cilindro", r);
    cin("Altura del cilindro", h);
    vol = 3.1416 * r ^ 2 * h;
    area = 2 * 3.1416 * r * h;
    cout "Area", area;
    cout "Volumen", vol;
}
