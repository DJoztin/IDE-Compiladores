    /* G3-2 (selectivo): A, B y C enteros diferentes, en forma descendente */
    int a, b, c;
    cin("Dato A (entero)", a);
    cin("Dato B (entero, diferente de A)", b);
    cin("Dato C (entero, diferente de A y B)", c);
    if (a == b || a == c || b == c) then
        cout "Los tres numeros deben ser diferentes";
    else
        if (a > b && a > c) then
            cout "Mayor", a;
            if (b > c) then cout "Medio", b; cout "Menor", c;
            else cout "Medio", c; cout "Menor", b; end;
        else
            if (b > a && b > c) then
                cout "Mayor", b;
                if (a > c) then cout "Medio", a; cout "Menor", c;
                else cout "Medio", c; cout "Menor", a; end;
            else
                cout "Mayor", c;
                if (a > b) then cout "Medio", a; cout "Menor", b;
                else cout "Medio", b; cout "Menor", a; end;
            end;
        end;
    end;
}
