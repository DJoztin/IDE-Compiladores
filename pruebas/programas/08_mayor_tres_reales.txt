main() {
    /* G3-3, G10-3 (selectivo): el mayor de tres reales (pueden ser iguales) */
    real a, b, c, m;
    cin("Primer real", a);
    cin("Segundo real", b);
    cin("Tercer real", c);
    if (a >= b && a >= c) then
        m = a;
    else
        if (b >= c) then
            m = b;
        else
            m = c;
        end;
    end;
    cout "El mayor es", m;
}
