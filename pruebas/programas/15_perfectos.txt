main() {
    /* G10-4 (repetitivo): numeros perfectos menores o iguales que N */
    int lim, n, d, suma;
    do {
        cin("N (entero >= 1)", lim);
        if (lim < 1) then
            cout "N debe ser 1 o mayor";
        end;
    } until (lim >= 1);
    n = 2;
    while (n <= lim) {
        suma = 0;
        d = 1;
        while (d < n) {
            if (n % d == 0) then
                suma = suma + d;
            end;
            d++;
        };
        if (suma == n) then
            cout "Numero perfecto", n;
        end;
        n++;
    };
}
