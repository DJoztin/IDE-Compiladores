main() {
    /* G8-3, G12-3 (selectivo): menor a $500 sin descuento; $500 a $1000: 5%;
       hasta $7000: 11%; hasta $15000: 18%; mayor a $15000: 25% */
    float monto, pagar;
    cin("Monto de la compra ($)", monto);
    if (monto < 0) then
        cout "El monto no puede ser negativo";
    else
        if (monto < 500) then
            pagar = monto;
        else
            if (monto <= 1000) then
                pagar = monto - monto * 0.05;
            else
                if (monto <= 7000) then
                    pagar = monto - monto * 0.11;
                else
                    if (monto <= 15000) then
                        pagar = monto - monto * 0.18;
                    else
                        pagar = monto - monto * 0.25;
                    end;
                end;
            end;
        end;
        cout "Total a pagar ($)", pagar;
    end;
}
