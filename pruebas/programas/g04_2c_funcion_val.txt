main() {
    /* G4-2: VAL = 100*V si NUM=1; 100^V si NUM=2; 100/V si NUM=3; 0 en otro caso */
    int num, v;
    float val;
    cin(num);
    cin(v);
    if (num == 1) then
        val = 100 * v;
    else
        if (num == 2) then
            val = 100 ^ v;
        else
            if (num == 3) then
                if (v != 0) then
                    val = 100.0 / v;
                else
                    val = 0;
                end;
            else
                val = 0;
            end;
        end;
    end;
    cout val;
}
