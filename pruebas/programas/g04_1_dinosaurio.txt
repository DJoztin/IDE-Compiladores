main() {
    /* G4-1 (secuencial): libras -> kilogramos ; pies -> metros */
    float libras, pies, kg, metros;
    cin("Peso del dinosaurio (libras)", libras);
    cin("Longitud del dinosaurio (pies)", pies);
    kg = libras * 0.4536;
    metros = pies * 0.3048;
    cout "Peso (kg)", kg;
    cout "Longitud (m)", metros;
}
