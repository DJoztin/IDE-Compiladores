main() {
    /* G3-5, G6-5, G7-5: N fabricas; por fabrica su CLAVE y la produccion
       de los 12 meses. Por fabrica imprime: clave, total anual y 1 si en
       julio supero 3 000 000 (0 si no). Al final: clave y total de la
       fabrica que mas produjo. */
    int n, i, clave, mes, cmax;
    float p, total, julio, tmax;
    cin(n);
    i = 1;
    while (i <= n) {
        cin(clave);
        total = 0;
        mes = 1;
        while (mes <= 12) {
            cin(p);
            total = total + p;
            if (mes == 7) then
                julio = p;
            end;
            mes++;
        };
        cout clave;
        cout total;
        if (julio > 3000000) then cout 1; else cout 0; end;
        if (i == 1) then
            cmax = clave;
            tmax = total;
        else
            if (total > tmax) then
                cmax = clave;
                tmax = total;
            end;
        end;
        i++;
    };
    cout cmax;
    cout tmax;
}
