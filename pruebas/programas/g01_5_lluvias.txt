main() {
    /* G1-5, G9-5, G11-5: por cada mes (1..12) se leen las lluvias de
       NORTE, CENTRO y SUR. Imprime: promedio anual del centro, mes y
       registro con menor lluvia en el sur, region con mayor lluvia anual
       (1 = norte, 2 = centro, 3 = sur). */
    int mes, mesmin, region;
    float n, c, s, tn, tc, ts, prom, smin;
    tn = 0; tc = 0; ts = 0;
    mes = 1;
    while (mes <= 12) {
        cin(n);
        cin(c);
        cin(s);
        tn = tn + n;
        tc = tc + c;
        ts = ts + s;
        if (mes == 1) then
            smin = s;
            mesmin = 1;
        else
            if (s < smin) then
                smin = s;
                mesmin = mes;
            end;
        end;
        mes++;
    };
    prom = tc / 12;
    cout prom;
    cout mesmin;
    cout smin;
    if (tn > tc && tn > ts) then
        region = 1;
    else
        if (tc > ts) then
            region = 2;
        else
            region = 3;
        end;
    end;
    cout region;
}
