main() {
    /* G1-5, G9-5, G11-5 (repetitivo): lluvias de NORTE, CENTRO y SUR
       en cada mes (1..12). Imprime: promedio anual del centro, mes y
       registro con menor lluvia en el sur, region con mayor lluvia anual
       (1 = norte, 2 = centro, 3 = sur). */
    int mes, mesmin, region;
    float n, c, s, tn, tc, ts, prom, smin;
    tn = 0; tc = 0; ts = 0;
    mes = 1;
    while (mes <= 12) {
        cout "Mes", mes;
        do {
            cin("Lluvia NORTE (>= 0)", n);
            if (n < 0) then
                cout "La lluvia no puede ser negativa";
            end;
        } until (n >= 0);
        do {
            cin("Lluvia CENTRO (>= 0)", c);
            if (c < 0) then
                cout "La lluvia no puede ser negativa";
            end;
        } until (c >= 0);
        do {
            cin("Lluvia SUR (>= 0)", s);
            if (s < 0) then
                cout "La lluvia no puede ser negativa";
            end;
        } until (s >= 0);
        tn = tn + n;
        tc = tc + c;
        ts = ts + s;
        if (mes == 1) then
            smin = s;
            mesmin = 1;
        else
            if (s < smin) then
                smin = s;
                mesmin = mes;
            end;
        end;
        mes++;
    };
    prom = tc / 12;
    cout "Promedio anual region CENTRO", prom;
    cout "Mes con menor lluvia en el SUR", mesmin;
    cout "Registro de ese mes", smin;
    if (tn > tc && tn > ts) then
        region = 1;
    else
        if (tc > ts) then
            region = 2;
        else
            region = 3;
        end;
    end;
    cout "Region con mayor lluvia (1 norte, 2 centro, 3 sur)", region;
}
