main() {
    /* G1-2, G7-2 (selectivo): positivo = 0, negativo = 1, nulo = 2 */
    int n;
    cin("Numero entero", n);
    if (n > 0) then
        cout "Positivo (0)", 0;
    else
        if (n < 0) then
            cout "Negativo (1)", 1;
        else
            cout "Nulo (2)", 2;
        end;
    end;
}
