main() {
    /* G8-3, G12-3: < 500 sin descuento; 500..1000 5%; ..7000 11%;
       ..15000 18%; mayor a 15000 25% */
    float monto, pagar;
    cin(monto);
    if (monto < 500) then
        pagar = monto;
    else
        if (monto <= 1000) then
            pagar = monto - monto * 0.05;
        else
            if (monto <= 7000) then
                pagar = monto - monto * 0.11;
            else
                if (monto <= 15000) then
                    pagar = monto - monto * 0.18;
                else
                    pagar = monto - monto * 0.25;
                end;
            end;
        end;
    end;
    cout pagar;
}
