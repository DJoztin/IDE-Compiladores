main() {
    /* G8-2, G12-2 (selectivo): 1 si P^3 + Q^4 - 2*P^2 < 680, 0 si no */
    int p, q;
    cin("Valor de P (entero)", p);
    cin("Valor de Q (entero)", q);
    if (p ^ 3 + q ^ 4 - 2 * p ^ 2 < 680) then
        cout "Satisface la expresion (1 = si)", 1;
    else
        cout "Satisface la expresion (0 = no)", 0;
    end;
}
