main() {
    /* G7-3 (selectivo): tres enteros en orden creciente */
    int a, b, c;
    cin("Primer entero", a);
    cin("Segundo entero", b);
    cin("Tercer entero", c);
    if (a <= b && a <= c) then
        cout "Menor", a;
        if (b <= c) then cout "Medio", b; cout "Mayor", c;
        else cout "Medio", c; cout "Mayor", b; end;
    else
        if (b <= a && b <= c) then
            cout "Menor", b;
            if (a <= c) then cout "Medio", a; cout "Mayor", c;
            else cout "Medio", c; cout "Mayor", a; end;
        else
            cout "Menor", c;
            if (a <= b) then cout "Medio", a; cout "Mayor", b;
            else cout "Medio", b; cout "Mayor", a; end;
        end;
    end;
}
