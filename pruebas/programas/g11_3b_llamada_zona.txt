main() {
    /* G11-3 (selectivo): costo de una llamada internacional segun la zona.
       12 America del Norte $2, 15 America Central $2.2,
       18 America del Sur $4.5, 19 Europa $3.5, 23 Asia $6,
       25 Africa $6, 29 Oceania $5 (por minuto) */
    int clave, minutos;
    float precio, costo;
    cin("Clave de zona (12, 15, 18, 19, 23, 25 o 29)", clave);
    cin("Minutos hablados (>= 1)", minutos);
    precio = 0;
    if (clave == 12) then precio = 2; end;
    if (clave == 15) then precio = 2.2; end;
    if (clave == 18) then precio = 4.5; end;
    if (clave == 19) then precio = 3.5; end;
    if (clave == 23 || clave == 25) then precio = 6; end;
    if (clave == 29) then precio = 5; end;
    if (precio == 0) then
        cout "Clave de zona no valida: usa 12, 15, 18, 19, 23, 25 o 29";
    else
        if (minutos < 1) then
            cout "Los minutos deben ser 1 o mas";
        else
            costo = precio * minutos;
            cout "Costo total de la llamada ($)", costo;
        end;
    end;
}
