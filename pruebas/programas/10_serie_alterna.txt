main() {
    /* G6-4 (repetitivo): 1 - 1/2 + 1/3 - 1/4 + ... +- 1/N */
    int n, i;
    float s, signo;
    do {
        cin("N (entero >= 1)", n);
        if (n < 1) then
            cout "N debe ser 1 o mayor";
        end;
    } until (n >= 1);
    s = 0;
    signo = 1;
    i = 1;
    while (i <= n) {
        s = s + signo / i;
        signo = -signo;
        i++;
    };
    cout "Resultado de la serie", s;
}
