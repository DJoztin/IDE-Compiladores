main() {
    /* G3-1, G10-1 (secuencial): perimetro y area de un rectangulo */
    float base, altura, per, area;
    cin("Base del rectangulo", base);
    cin("Altura del rectangulo", altura);
    per = 2 * (base + altura);
    area = base * altura;
    cout "Perimetro", per;
    cout "Area", area;
}
