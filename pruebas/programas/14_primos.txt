main() {
    /* G9-4: obtiene y cuenta los primos menores que M */
    int m, n, d, primo, cuenta;
    cin(m);
    cuenta = 0;
    n = 2;
    while (n < m) {
        primo = 1;
        d = 2;
        while (d * d <= n && primo == 1) {
            if (n % d == 0) then
                primo = 0;
            end;
            d++;
        };
        if (primo == 1) then
            cout n;
            cuenta++;
        end;
        n++;
    };
    cout cuenta;
}
