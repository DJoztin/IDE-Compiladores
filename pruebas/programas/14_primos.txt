main() {
    /* G9-4 (repetitivo): obtiene y cuenta los primos menores que M */
    int m, n, d, primo, cuenta;
    do {
        cin("M (entero >= 2)", m);
        if (m < 2) then
            cout "M debe ser 2 o mayor";
        end;
    } until (m >= 2);
    cuenta = 0;
    n = 2;
    while (n < m) {
        primo = 1;
        d = 2;
        while (d * d <= n && primo == 1) {
            if (n % d == 0) then
                primo = 0;
            end;
            d++;
        };
        if (primo == 1) then
            cout "Primo", n;
            cuenta++;
        end;
        n++;
    };
    cout "Cantidad de primos menores que M", cuenta;
}
